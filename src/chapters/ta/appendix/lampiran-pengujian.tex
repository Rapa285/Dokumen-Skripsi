\chapter{Kode Pengujian}
\label{app:kode-pengujian}

Lampiran ini memuat seluruh kode yang digunakan dalam pengujian nonfungsional
(P-KNF-01 hingga P-KNF-08) dan pengujian integrasi (P-INT-01). Kode diorganisasi
mulai dari fungsi pendukung, implementasi fungsi verifikasi yang diuji, kemudian
setiap kasus pengujian secara berurutan. Seluruh kode berada pada berkas
\texttt{audit-trail/src/tests.rs}, kecuali \texttt{verify\_record} dan
\texttt{verify\_chain} yang berada pada \texttt{audit-trail/src/utils.rs}.

\begin{minted}[linenos, breaklines]{rust}
use aes_gcm::{aead::Aead, Aes256Gcm, KeyInit, Nonce};
use base64::{engine::general_purpose::STANDARD, Engine as _};
use iota_types::base_types::IotaAddress;
use iota_types::crypto::{get_key_pair, EncodeDecodeBase64, IotaKeyPair, Signature as IotaSignature};
use p256::pkcs8::{EncodePrivateKey, EncodePublicKey, LineEnding};
use rand::rngs::OsRng;
use shared_crypto::intent::{Intent, IntentMessage};
 
struct AlsKeys {
    private_pem: String,
    public_pem: String,
}
 
fn generate_als_keys() -> AlsKeys {
    let secret = p256::SecretKey::random(&mut OsRng);
    AlsKeys {
        private_pem: secret.to_pkcs8_pem(LineEnding::LF).unwrap().to_string(),
        public_pem: secret.public_key().to_public_key_pem(LineEnding::LF).unwrap(),
    }
}
 
fn generate_sender() -> IotaKeyPair {
    IotaKeyPair::Ed25519(get_key_pair().1)
}
 
fn address_of(key_pair: &IotaKeyPair) -> IotaAddress {
    IotaAddress::from(&key_pair.public())
}
 
/// Meniru langkah 1-5 ALSClient::build_and_send tanpa pengiriman jaringan
fn seal_event(event: &AuditEvent, sender: &IotaKeyPair, als_public_pem: &str) -> EncryptedSignedEvent {
    let encrypted = aes_encrypt(&serde_json::to_vec(event).unwrap()).unwrap();
    let intent_message = IntentMessage::new(Intent::personal_message(), encrypted.ciphertext.clone());
 
    EncryptedSignedEvent {
        enc_aes_key: ecies_encrypt_key(&encrypted.key, als_public_pem).unwrap(),
        ciphertext: STANDARD.encode(&encrypted.ciphertext),
        nonce: STANDARD.encode(&encrypted.nonce),
        signature: IotaSignature::new_secure(&intent_message, sender).encode_base64(),
        iota_address: address_of(sender).to_string(),
    }
}
 
/// Membuka isi event dengan kunci privat ALS, sebagaimana dilakukan saat pembacaan log
fn open_event(sealed: &EncryptedSignedEvent, als_private_pem: &str) -> Result<AuditEvent, String> {
    let session_key = ecies_decrypt_key(&sealed.enc_aes_key, als_private_pem)?;
    let nonce = STANDARD.decode(&sealed.nonce).map_err(|e| e.to_string())?;
    let ciphertext = STANDARD.decode(&sealed.ciphertext).map_err(|e| e.to_string())?;
    if nonce.len() != 12 {
        return Err("panjang nonce tidak valid".to_string());
    }
 
    let plaintext = Aes256Gcm::new_from_slice(&session_key)
        .map_err(|e| e.to_string())?
        .decrypt(Nonce::from_slice(&nonce), ciphertext.as_slice())
        .map_err(|_| "dekripsi AES-GCM gagal".to_string())?;
 
    serde_json::from_slice(&plaintext).map_err(|e| e.to_string())
}
 
fn flip_first_byte(base64_value: &str) -> String {
    let mut bytes = STANDARD.decode(base64_value).unwrap();
    bytes[0] ^= 0x01;
    STANDARD.encode(bytes)
}

#[test]
fn encrypted_event_tampered_ciphertext_rejected() {
    let als = generate_als_keys();
    let mut sealed = seal_event(&make_ev4_event(), &generate_sender(), &als.public_pem);
    sealed.ciphertext = flip_first_byte(&sealed.ciphertext);
 
    assert!(
        Utils::verify_event_signature(&sealed).is_err(),
        "Ciphertext yang dimodifikasi harus ditolak"
    );
}
 
#[test]
fn encrypted_event_tampered_nonce_fails_on_decrypt() {
    let als = generate_als_keys();
    let mut sealed = seal_event(&make_ev4_event(), &generate_sender(), &als.public_pem);
    sealed.nonce = flip_first_byte(&sealed.nonce);
 
    // nonce tidak ditandatangani, sehingga lolos pada penerimaan
    assert!(Utils::verify_event_signature(&sealed).is_ok());
    // tetapi authentication tag AES-GCM menolaknya saat pembacaan
    assert!(
        open_event(&sealed, &als.private_pem).is_err(),
        "Nonce yang dimodifikasi harus gagal didekripsi"
    );
}

#[test]
fn stored_record_hides_event_content() {
    let als = generate_als_keys();
    let original_event = make_ev4_event();
    let sealed = seal_event(&original_event, &generate_sender(), &als.public_pem);
 
    let record = create_audit_record(sealed, None);
    let json_line = serde_json::to_string(&record).unwrap();
 
    let sensitive_values = [
        original_event.event.actor_id.as_str(),
        original_event.event.target_object.as_str(),
    ];
    assert!(
        sensitive_values.iter().all(|value| !json_line.contains(value)),
        "Record pada berkas log tidak boleh memuat isi event dalam plaintext"
    );
}
 
#[test]
fn stored_record_readable_only_with_als_key() {
    let als = generate_als_keys();
    let other_key = generate_als_keys();
    let original_event = make_ev4_event();
    let sealed = seal_event(&original_event, &generate_sender(), &als.public_pem);
    let record = create_audit_record(sealed, None);
 
    assert!(open_event(&record.event, &other_key.private_pem).is_err());
 
    let opened = open_event(&record.event, &als.private_pem).expect("kunci ALS harus berhasil");
    assert_eq!(
        serde_json::to_value(&opened).unwrap(),
        serde_json::to_value(&original_event).unwrap(),
        "Isi hasil dekripsi harus identik dengan event asli"
    );
}
